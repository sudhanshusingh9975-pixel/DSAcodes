# Program to Sort Online Orders using Merge Sort

# Function to merge two sorted lists
def merge(left, right):

    result = []
    i = 0
    j = 0

    while i < len(left) and j < len(right):

        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1

    # Add remaining elements
    result.extend(left[i:])
    result.extend(right[j:])

    return result


# Merge Sort Function
def merge_sort(orders):

    if len(orders) <= 1:
        return orders

    # Divide the list into two halves
    mid = len(orders) // 2

    left = merge_sort(orders[:mid])
    right = merge_sort(orders[mid:])

    # Merge the sorted halves
    return merge(left, right)


# Main Program
print("ONLINE ORDER DELIVERY SYSTEM")

# Delivery times in minutes
orders = [45, 20, 60, 15, 35, 10, 50]

print("\nOriginal Delivery Times:")
print(orders)

# Sort orders using Merge Sort
sorted_orders = merge_sort(orders)

print("\nSorted Delivery Times (Fastest First):")
print(sorted_orders)

print("\nOrder Priority:")

for i in range(len(sorted_orders)):
    print("Order", i + 1, ":", sorted_orders[i], "minutes")
