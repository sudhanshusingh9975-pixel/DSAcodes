#include <iostream>
using namespace std;

int main()
{
    int n, key, choice;

    cout << "Enter number of elements: ";
    cin >> n;

    int arr[100];

    cout << "Enter array elements: ";
    for(int i = 0; i < n; i++)
    {
        cin >> arr[i];
    }

    do
    {
        cout << "\n----- SEARCHING MENU -----";
        cout << "\n1. Linear Search";
        cout << "\n2. Binary Search";
        cout << "\n0. Exit";

        cout << "\nEnter your choice: ";
        cin >> choice;

        switch(choice)
        {
            case 1:
            {
                cout << "Enter element to search: ";
                cin >> key;

                int found = 0;

                for(int i = 0; i < n; i++)
                {
                    if(arr[i] == key)
                    {
                        cout << "Element found at position "
                             << i + 1 << endl;
                        found = 1;
                        break;
                    }
                }

                if(found == 0)
                    cout << "Element not found!" << endl;

                break;
            }

            case 2:
            {
                // Sort array in ascending order
                for(int i = 0; i < n - 1; i++)
                {
                    for(int j = 0; j < n - i - 1; j++)
                    {
                        if(arr[j] > arr[j + 1])
                        {
                            int temp = arr[j];
                            arr[j] = arr[j + 1];
                            arr[j + 1] = temp;
                        }
                    }
                }

                cout << "Sorted array: ";
                for(int i = 0; i < n; i++)
                    cout << arr[i] << " ";

                cout << "\nEnter element to search: ";
                cin >> key;

                int low = 0, high = n - 1;
                int found = 0;

                while(low <= high)
                {
                    int mid = (low + high) / 2;

                    if(arr[mid] == key)
                    {
                        cout << "Element found at position "
                             << mid + 1 << endl;
                        found = 1;
                        break;
                    }
                    else if(key < arr[mid])
                    {
                        high = mid - 1;
                    }
                    else
                    {
                        low = mid + 1;
                    }
                }

                if(found == 0)
                    cout << "Element not found!" << endl;

                break;
            }

            case 0:
                cout << "Program terminated.";
                break;

            default:
                cout << "Invalid choice!";
        }

    } while(choice != 0);

    return 0;
}
